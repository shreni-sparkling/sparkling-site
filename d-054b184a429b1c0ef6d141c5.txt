async function storeFile(SB2, ref, data, name) {
  const m = /^data:([^;]+);base64,(.*)$/.exec(String(data || ""));
  if (!m || String(data).length > 9e6) return "";
  if (!/^(application\/pdf|image\/(png|jpeg))$/.test(m[1])) return "";
  const bytes = Uint8Array.from(atob(m[2]), (ch) => ch.charCodeAt(0));
  const path = `orders/${ref}/${Date.now().toString(36)}-${name.replace(/[^a-z0-9.\-]+/gi, "-").slice(0, 80)}`;
  const { error } = await SB2.storage.from("sj-files").upload(path, bytes, { contentType: m[1], upsert: true });
  if (error) return "";
  return path;

function splitRow(line, sep) {
  if (sep === "	") return line.split("	");
  const out = [];
  let cur = "", q = false;
  for (let i = 0; i < line.length; i++) {
    const ch = line[i];
    if (q) {
      if (ch === '"' && line[i + 1] === '"') {
        cur += '"';
        i++;
      } else if (ch === '"') q = false;
      else cur += ch;
    } else if (ch === '"') q = true;
    else if (ch === sep) {
      out.push(cur);
      cur = "";
    } else cur += ch;
  }
  out.push(cur);
  return out;
}
async function customerImport(SB2, who, text) {
  if (!text.trim()) return { ok: false, error: "Paste the rows (with the header row) or pick a CSV file." };

  const lines = text.replace(/\r\n?/g, "\n").split("\n").filter((l) => l.trim());
  const sep = lines[0].includes("	") ? "	" : lines[0].includes(";") && !lines[0].includes(",") ? ";" : ",";

async function opsSettings(SB2, who, u, b, emailOnly = false) {
  const owner = isOwner(u);
  const ops2 = await getS(SB2, "ops");
  const changed = [];
  const keys = emailOnly ? [...EMAIL_KEYS, ...owner ? OWNER_EMAIL_KEYS : []] : [...OPS_KEYS, ...owner ? OWNER_OPS_KEYS : []];
  if (!owner && (emailOnly ? OWNER_EMAIL_KEYS : OWNER_OPS_KEYS).some((k) => b[k] !== void 0)) return { ok: false, error: "Only an owner can change pricing, programs, payment / return details, security alerts, the email footer, sender or logo, mail routing or site settings." };
  if (!owner && !emailOnly && b.tpl && typeof b.tpl === "object" && Object.keys(b.tpl).some(ownerOnlyTpl)) return { ok: false, error: "Only an owner can change sign-in, receipt or staff emails \u2014 they go out without review." };

  for (const k of keys) if (b[k] !== void 0 && JSON.stringify(b[k]).length < 6e4) {
    ops2[k] = cleanSetting(k, b[k]);
    changed.push(k);
  }
  if (b.rules && typeof b.rules === "object" && !emailOnly) {
    ops2.rules = { ...ops2.rules || {}, ...b.rules };
    changed.push("rules");
  }
  if (changed.length) await putS(SB2, "ops", ops2, who);
  if (!emailOnly && b.tpl && typeof b.tpl === "object") {
    const T = await getS(SB2, "tpl");
    for (const [k, v] of Object.entries(b.tpl)) {
      if (!/^[a-z0-9_]{2,40}$/.test(k) || !v || typeof v !== "object") continue;
      if (!owner && ownerOnlyTpl(k)) return { ok: false, error: "Only an owner can change the \u201C" + k + "\u201D email \u2014 it goes out without review." };
      await SB2.from("audit").insert({ user_name: who, action: "tpl_saved", target: k, details: JSON.stringify(T[k] || null).slice(0, 4e3) });
      T[k] = { subject: str(v.subject, 300), body: String(v.body || "").slice(0, 2e4) };
    }
    await putS(SB2, "tpl", T, who);
    changed.push("tpl");
  }
  if (!emailOnly && b.tplReset) {

function laToday() {
  const p = new Intl.DateTimeFormat("en-US", { timeZone: LA, year: "numeric", month: "2-digit", day: "2-digit" }).formatToParts(/* @__PURE__ */ new Date());
  const g = (t) => +p.find((x) => x.type === t).value;
  return new Date(Date.UTC(g("year"), g("month") - 1, g("day")));
}
var iso = (d) => d.toISOString().slice(0, 10);
var addDays = (d, n) => new Date(d.getTime() + n * 864e5);
function parseDue(phrase) {
  const t = String(phrase || "").toLowerCase().replace(/[.,]/g, " ").replace(/\s+/g, " ").trim();
  if (!t) return { date: "", how: "" };
  const today = laToday();
  const dow = today.getUTCDay();
  const pick = (d, how) => ({ date: iso(d), how });
  if (/\b(asap|rush|right away|today)\b/.test(t)) return pick(today, "today (rush)");
  if (/\btomorrow\b/.test(t)) return pick(addDays(today, 1), "tomorrow");
  let m = /\bin (\d+|a|an|one|two|three|four|five|six|seven|ten) (day|week|business day)s?\b/.exec(t);
  if (m) {
    const n = { a: 1, an: 1, one: 1, two: 2, three: 3, four: 4, five: 5, six: 6, seven: 7, ten: 10 }[m[1]] ?? +m[1];
    if (m[2] === "week") return pick(addDays(today, 7 * n), `in ${n} week${n > 1 ? "s" : ""}`);
    if (m[2] === "business day") {
      let d = today, c = 0;
      while (c < n) {
        d = addDays(d, 1);
        if (d.getUTCDay() % 6) c++;
      }
      return pick(d, `in ${n} business days`);
    }
    return pick(addDays(today, n), `in ${n} day${n > 1 ? "s" : ""}`);
  }
  if (/\bend of (the )?(this )?week\b/.test(t)) return pick(addDays(today, (5 - dow + 7) % 7 || 7), "end of this week (Friday)");
  if (/\bnext week\b/.test(t)) return pick(addDays(today, (5 - dow + 7) % 7 + 7), "next week (Friday)");
  if (/\bend of (the )?month\b/.test(t)) {
    const d = new Date(Date.UTC(today.getUTCFullYear(), today.getUTCMonth() + 1, 0));
    return pick(d, "end of the month");
  }
  m = /\b(next|this|by|on|before|due)?\s*(sunday|monday|tuesday|wednesday|thursday|friday|saturday|sun|mon|tue|tues|wed|thu|thur|thurs|fri|sat)\b/.exec(t);
  if (m) {
    const key = m[2].slice(0, 3);
    const target = DOW.findIndex((x) => x.startsWith(key));
    let diff = (target - dow + 7) % 7;
    if (diff === 0) diff = 7;
    if (m[1] === "next" && diff <= 2) diff += 7;
    return pick(addDays(today, diff), (m[1] === "next" ? "next " : "this ") + DOW[target]);
  }
  m = /\b(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)[a-z]*\.? (\d{1,2})(?:st|nd|rd|th)?\b/.exec(t) || /\b(\d{1,2})(?:st|nd|rd|th)? (?:of )?(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)[a-z]*\b/.exec(t);
  if (m) {
    const mi = MON.indexOf(isNaN(+m[1]) ? m[1] : m[2]);
    const day = +(isNaN(+m[1]) ? m[2] : m[1]);
    let d = new Date(Date.UTC(today.getUTCFullYear(), mi, day));
    if (d < today) d = new Date(Date.UTC(today.getUTCFullYear() + 1, mi, day));
    return pick(d, `${MON[mi]} ${day}`);
  }
  m = /\b(?:by |on |before )?the (\d{1,2})(?:st|nd|rd|th)\b/.exec(t);
  if (m) {
    const day = +m[1];
    let d = new Date(Date.UTC(today.getUTCFullYear(), today.getUTCMonth(), day));
    if (d <= today) d = new Date(Date.UTC(today.getUTCFullYear(), today.getUTCMonth() + 1, day));

  await SB.from("sessions").insert({ token_hash: await sha256(token), user_name: name, epoch: u.epoch || 0, limited: limitedS, device: deviceOf(device) });
  return token;
}
async function sessionGet(token) {
  if (!token) return null;
  const { data: s } = await SB.from("sessions").select("*").eq("token_hash", await sha256(String(token))).maybeSingle();
  if (!s || new Date(s.expires_at).getTime() < Date.now()) return null;
  const { data: u } = await SB.from("team_users").select("*").eq("name", s.user_name).maybeSingle();
  if (!u || !u.active || (u.epoch || 0) !== (s.epoch || 0)) return null;
  return { s, u };
}

}

async function withAuth(ctx, fn, opt = {}) {
  const g = await sessionGet(ctx.body.token);
  if (!g) return { ok: false, auth: false, error: "Session expired. Log in again." };
  const { s, u } = g;
  if (u.must_change && !opt.allowLimited) return { ok: false, mustChange: true, error: "First sign-in: add your recovery email and choose a new password." };
  if (!u.email && u.role !== "owner" && !opt.allowLimited) return { ok: false, mustEmail: true, error: "Add your email first so you can reset a lost password." };
  if (s.limited && !opt.allowLimited) return { ok: false, mustEnroll2fa: true, error: "Set up 2-step verification first." };
  if (opt.owner && u.role !== "owner") return { ok: false, error: "Only an owner can do this." };
  if (opt.perm && u.role !== "owner" && !(u.perms || []).includes(opt.perm) && !(opt.sales && (u.perms || []).includes("sales")) && !(opt.proc && (u.perms || []).includes("processing"))) return { ok: false, forbidden: true, error: `Your account doesn\u2019t have access to ${opt.perm}.` };
  const r = await fn(s.user_name, u, g);
  if (r && typeof r === "object") r.user = s.user_name;
  return r;
}
async function login2(ctx) {
  const b = ctx.body;
  const name = cleanName(b.user);
  if (await throttle(SB, "login:" + name, 8, 15) || await throttle(SB, "login:ip:" + (ctx.ip || "?"), 30, 15) || await throttle(SB, "login:*", 120, 15)) return { ok: false, error: "Too many attempts. Try again in 15 minutes." };
  const { data: u } = await SB.from("team_users").select("*").eq("name", name).maybeSingle();
  const passOk = await checkHash(u && u.active ? u.pw_hash : null, String(b.pass || ""));
  if (!u || !u.active || !u.pw_hash || !passOk) {
    await audit2(name, "login_failed", "", "password");
    await loginLog(name, "wrong password", b.device, "password", ctx.ip);
    return { ok: false, error: "Wrong username or password. New login? Use the setup link or code an owner sent you." };
  }
  const sec = await setting("security", {});
  const need2 = !!u.email && (!!u.totp_on || !!sec.force2fa);
  let how = "password";
  if (need2) {
    const otp = String(b.otp || "").trim().toUpperCase();
    if (!otp) {
      if (await throttle(SB, "a2s:send:" + name, 5, 15)) return { ok: false, needOtp: true, error: "A code is already on its way to " + maskEmail(u.email) + " \u2014 enter it below." };
      const { code, minutes } = await issueCode(SB, "admin_2step", name);
      try {
        await sendTpl(SB, "staff_code", u.email, { name, code, minutes }, { by: "automatic" });
      } catch {
        return { ok: false, error: "Couldn\u2019t send your sign-in code. Ask an owner." };
      }
      return { ok: false, needOtp: true, error: "We emailed a 6-digit code to " + maskEmail(u.email) + ". Enter it below (valid 10 minutes)." };
    }
    if (/^[A-Z0-9]{5}-[A-Z0-9]{5}$/.test(otp)) {
      const h = await hashSecret(otp);
      const { data: left, error: re } = await SB.rpc("recovery_use", { p_name: name, p_hash: h });
      if (re) console.error("recovery_use rpc failed (apply migration_v6.sql)", re.message);
      if (re || left == null) {
        await fail("login:" + name);
        await loginLog(name, "wrong code", b.device, "recovery", ctx.ip);
        return { ok: false, needOtp: true, error: "That recovery code didn\u2019t work." };
      }
      how = "recovery code";
      await ownerAlert(name, "Signed in with a recovery code", Number(left) + " left");
    } else {
      const r = await useCode(SB, "admin_2step", name, otp);
      if (!r.ok) {
        await fail("login:" + name);
        await loginLog(name, "wrong code", b.device, "email code", ctx.ip);
        return { ok: false, needOtp: true, error: r.error };
      }
      how = "password + email code";
    }
    if (!u.totp_on) {
      await SB.from("team_users").update({ totp_on: true }).eq("name", name);
      u.totp_on = true;
    }
  }
  const dv = deviceOf(b.device);
  const key = dv.device_id || dv.label + "|" + dv.screen;
  const seen = u.devices || [];
  const upd = { last_login: (/* @__PURE__ */ new Date()).toISOString() };
  if (!seen.includes(key)) {
    upd.devices = [...seen, key].slice(-30);
    if (seen.length) await ownerAlert(name, SEC_ALERT.new_device, [dv.label, dv.screen, dv.tz, ctx.ip].filter(Boolean).join(" \xB7 "));
  }
  await SB.from("team_users").update(upd).eq("name", name);
  await clearLimit(SB, "login:" + name);
  await audit2(name, "login", "", how);
  await loginLog(name, "ok", b.device, how, ctx.ip);
  return payload(name, u, b.device);
}

async function userAdmin(who, b) {
  const action = String(b.action || "");
  const name = cleanName(b.username || b.user || b.name);
  const p = b.patch && typeof b.patch === "object" ? b.patch : { ...b.perms !== void 0 ? { perms: b.perms } : {}, ...b.role !== void 0 ? { role: b.role } : {}, ...b.active !== void 0 ? { active: b.active } : {}, ...b.email !== void 0 ? { email: b.email } : {} };
  const owners = async () => ((await SB.from("team_users").select("name").eq("role", "owner").eq("active", true)).data || []).map((x) => x.name);
  if (action === "create") {
    if (!name || name.length < 2) return { ok: false, error: "Username, please (letters, numbers, . _ -)." };
    const { data: ex } = await SB.from("team_users").select("name").eq("name", name).maybeSingle();
    if (ex) return { ok: false, error: "That username exists." };
    const perms = (Array.isArray(p.perms) ? p.perms : []).filter((x) => PERMS.includes(x));
    const role = p.role === "owner" ? "owner" : "staff";
    const email = String(b.email || p.email || "").trim().toLowerCase();
    if (!okMail(email)) return { ok: false, error: "Add their work email \u2014 the setup link is emailed to them (or create the login, then set a starter password)." };
    await SB.from("team_users").insert({ name, role, perms, email: email || null, created_by: who, must_change: false });
    if (b.requestId) await SB.from("access_requests").update({ status: "Done", decided_by: who }).eq("id", b.requestId);
    const setup = await issueSetup(who, name, email);
    await audit2(who, "user_created", name, role + " \xB7 " + perms.join(",") + (b.displayName ? " \xB7 " + String(b.displayName).slice(0, 60) : "") + " \xB7 setup code " + (setup.emailed ? "emailed" : "shown"));
    await ownerAlert(who, SEC_ALERT.user_created, name);
    return { ...await usersList(), ...setup };
  }
  const { data: cur } = await SB.from("team_users").select("*").eq("name", name).maybeSingle();
  if (!cur) return { ok: false, error: "No such login." };
  if (action === "update") {
    const upd = {};
    if (p.perms !== void 0) upd.perms = (Array.isArray(p.perms) ? p.perms : []).filter((x) => PERMS.includes(x));
    if (p.role !== void 0) upd.role = p.role === "owner" ? "owner" : "staff";
    if (p.active !== void 0) upd.active = !!p.active;
    if (p.email !== void 0) {
      const e = String(p.email || "").trim().toLowerCase();
      if (e && !okMail(e)) return { ok: false, error: "That email doesn\u2019t look right." };
      upd.email = e || null;
    }
    if (!Object.keys(upd).length) return { ok: false, error: "Nothing to change." };
    if (name === who && (upd.active === false || upd.role === "staff")) return { ok: false, error: "You can\u2019t deactivate or demote yourself." };
    if (cur.role === "owner" && (upd.active === false || upd.role === "staff") && (await owners()).length <= 1) return { ok: false, error: "Keep at least one active owner." };
    if (name !== who && (upd.active === false || upd.role || upd.perms)) upd.epoch = (cur.epoch || 0) + 1;
    await SB.from("team_users").update(upd).eq("name", name);
    if (upd.epoch) await SB.from("sessions").delete().eq("user_name", name);
    await audit2(who, "user_updated", name, JSON.stringify({ ...upd, epoch: void 0 }));
    await ownerAlert(who, SEC_ALERT.user_updated, name + " \xB7 " + Object.keys(upd).filter((k) => k !== "epoch").join(", "));
    return usersList();
  }
  if (action === "reset") {
    if (!cur.email || !okMail(cur.email)) return { ok: false, error: "This login has no email \u2014 add one first, or set a starter password instead." };
    const epoch = (cur.epoch || 0) + 1;
    await SB.from("team_users").update({ pw_hash: null, must_change: false, epoch }).eq("name", name);
    await SB.from("sessions").delete().eq("user_name", name);
    const setup = await issueSetup(who, name, String(cur.email));
    await audit2(who, "password_reset_issued", name, "setup code " + (setup.emailed ? "emailed" : "shown") + " by " + who);
    await ownerAlert(who, "Password reset for " + name, "");
    return { ...await usersList(), ...setup };
  }
  if (action === "delete") {
    if (name === who) return { ok: false, error: "You can\u2019t delete yourself." };
    if (cur.role === "owner" && (await owners()).length <= 1) return { ok: false, error: "Keep at least one active owner." };
    await SB.from("team_users").delete().eq("name", name);
    await audit2(who, "user_deleted", name, "");
    await ownerAlert(who, SEC_ALERT.user_deleted, name);
    return usersList();
  }
  if (action === "revoke") {
    await SB.from("team_users").update({ epoch: (cur.epoch || 0) + 1 }).eq("name", name);
    await SB.from("sessions").delete().eq("user_name", name);
    await audit2(who, "sessions_revoked", name, "");
    return usersList();
  }
  if (action === "starter") {
    const starter = String(b.starter || "");
    const bad = checkPass(starter);
    if (bad) return { ok: false, error: bad };
    await SB.from("team_users").update({ pw_hash: await newHash(starter), must_change: true, epoch: (cur.epoch || 0) + 1 }).eq("name", name);
    await SB.from("sessions").delete().eq("user_name", name);
    await audit2(who, "password_reset_issued", name, "starter password set by " + who);
    return usersList();
  }
  return { ok: false, error: "Unknown action." };
}
async function bootstrap(b, ip) {

  const tok = String(data?.value?.token || "");

async function readCapped(req, cap3) {
  if (!req.body) return "";
  const rd = req.body.getReader();
  const parts = [];
  let n = 0;
  for (; ; ) {
    const { done, value } = await rd.read();
    if (done) break;
    n += value.byteLength;
    if (n > cap3) {
      try {
        await rd.cancel();
      } catch {
      }
      return null;
    }
    parts.push(value);
  }
  const all = new Uint8Array(n);
  let o = 0;
  for (const p of parts) {
    all.set(p, o);
    o += p.byteLength;
  }
  return new TextDecoder().decode(all);
}
